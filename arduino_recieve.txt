#include <Wire.h>

const byte I2C_ADDRESS = 0x08;

void receiveEvent(int numBytes) {
  char message[33];  // 32 payload bytes + null terminator
  int index = 0;

  while (Wire.available() && index < 32) {
    message[index++] = (char)Wire.read();
  }

  message[index] = '\0';

  Serial.print("Received: ");
  Serial.println(message);
}

void setup() {
  Serial.begin(115200);

  Wire.begin(I2C_ADDRESS);
  Wire.onReceive(receiveEvent);
}

void loop() {
}
